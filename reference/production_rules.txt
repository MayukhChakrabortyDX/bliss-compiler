//@ means these are token names.

IMPORT -> "import" IDENTIFIER ("." (IDENTIFIER | "(" IDENTIFIER*[separated_by ','] ")") )* ';' ;
USING -> "using" IDENTIFIER ("." (IDENTIFIER | "(" IDENTIFIER*[separated_by ','] ")") )* ';' ;

PROGRAM -> (IMPORT | USING)* (ANNOTATION? (FUNC_DECL | DATA_DECL | ACTION_DECL | BIND_DECL))*;

BIND_DECL -> "bind" @IDENTIFIER "with" ( @IDENTIFIER | ('(' @IDENTIFIER (',' @IDENTIFIER)* ')') ) "as" @IDENTIFIER '{'

    FUNC_DECL (FUNC_DECL)*

'}'

BUILTIN_TYPES -> `${u, i}${8, 16, 32, 64}` | `f${32, 64}` | 'ptr';

COMPOSITE_TYPE -> @IDENTIFIER '::' (
    '(' ( @IDENTIFIER (',' @IDENTIFIER)* ) ')'
    | @IDENTIFIER
);

TYPE -> BUILTIN_TYPES | @IDENTIFIER | COMPOSITE_TYPE | '#' TYPE | '[' TYPE ']' | '`' TYPE | '$' TYPE | TYPE '[' @INTEGER ']' ;

ACTION_DECL -> "action" @IDENTIFIER '{' REDUCED_FUNC_DECL ';' (REDUCED_FUNC_DECL ';')* '}';

DATA_SOLO -> "data" @IDENTIFIER ';';

DATA_FIELD -> TYPE @IDENTIFIER ';';

DATA_STRUCT_DECL -> "data" @IDENTIFIER "{" DATA_FIELD (DATA_FIELD)* "}"; //one is required atleast

DATA_SCALAR_DECL -> "data" @IDENTIFIER "(" TYPE ")" ';' ;
DATA_ARRAY_DECL -> "data" @IDENTIFIER "[" TYPE ',' @INTEGER "]" ';';

DATA_DECL -> DATA_SOLO | DATA_STRUCT_DECL | DATA_SCALAR_DECL | DATA_ARRAY_DECL;

NUMBER -> @REAL_NUMBER | @INTEGER

ATOMIC_EXPRESSION -> @IDENTIFIER | NUMBER | STRING | "(" EXPRESSION ")" | "[" EXPRESSION ( "|" EXPRESSION )? "]" | 
"#[" EXPRESSION ( "|" EXPRESSION )? "]" | '`' ATOMIC_EXPRESSION | 'adrs' ATOMIC_EXPRESSION | `sizeof` ATOMIC_EXPRESSION |
'$' EXPRESSION | CALL_SIGNATURE ;

CALL_SIG -> ATOMIC_EXPRESSION ("(" (EXPRESSION ("," EXPRESSION)*)? ")")?;

BINDING_ATTACHMENT -> CALL_SIG ("::" CALL_SIG)?;
MAGNETIC_CALL -> BINDING_ATTACHMENT ("->" BINDING_ATTACHMENT)*;
MEMBER_ACCESS -> MAGNETIC_CALL ("." MAGNETIC_CALL)* ;
DIVIDE_MULTIPLY -> MEMBER_ACCESS (('/' | '*') MEMBER_ACCESS)*;
ADD_SUB_OPS -> DIVIDE_MULTIPLY (('+' | '-') DIVIDE_MULTIPLY)*;
GTE_LTE_GTHN_LTHN_OPS -> ADD_SUB_OPS (('>' | '<' | '>=' | '<=') ADD_SUB_OPS)?;
EQUALITY -> GTE_LTE_GTHN_LTHN_OPS (('==' | '!=') EQUALITY)*;
ASSIGNMENT -> GTE_LTE_GTHN_LTHN_OPS ('=' GTE_LTE_GTHN_LTHN_OPS)?;

EXPRESSION -> ASSIGNMENT;

LOOP -> 'loop' IDENTIFIER? '{' STATEMENT* '}';
CONDITION -> 'if' '(' EXPRESSION ')' '{' STATEMENT* '}' ('elif' '(' EXPRESSION ')' '{' STATEMENT* '}')* ('else' '{' STATEMENT* '}')? ;

RETURN -> "return" EXPRESSION? ";";
BREAK -> "break" IDENTIFIER? ";";

ALIAS_STATEMENT -> 'alias' TYPE 'as' @IDENTIFIER ';';
STATEMENT -> TRANSFORM_DECL | TRANSFORM_STATEMENT | RETURN | LOOP | BREAK | CONDITION | (CALL_SIGNATURE ';') | VARIABLE_DECL | ASSIGNMENT | VIEW_STATEMENT | ALIAS_STATEMENT | EXPRESSION | PTR_DECL;

ARGUMENT_NAME -> @IDENTIFIER;

TYPE_BASED_ARGUMENT -> TYPE ARGUMENT_NAME;
ACTION_BASED_ARGUMENT -> ARGUMENT_NAME 'with' (
    '(' @IDENTIFIER (',' @IDENTIFIER)* ')'
    | @IDENTIFIER
);

//! we will expand this later - the @IDENTIFIER will be a strict data once more are available
DATA_ACTION_BASED_ARGUMENT -> @IDENTIFIER ACTION_BASED_ARGUMENT; //one more identifier because of the strict data type

ARGUMENT_LIST -> TYPE_BASED_ARGUMENT | ACTION_BASED_ARGUMENT | DATA_ACTION_BASED_ARGUMENT;

ANNOTATION -> '@' @IDENTIFIER;

FUNC_DECL -> "fx" @IDENTIFIER "(" ARGUMENT_LIST*[separated_by ','] ")" ':' TYPE "{" STATEMENT* "}";
REDUCED_FUNC_DECL -> "fx" @IDENTIFIER "(" ARGUMENT_LIST*[separated_by ','] ")" ':' TYPE;

//variable declaration
VARIABLE_DECL -> "unsafe"? "let" @IDENTIFIER ":" TYPE "=" EXPRESSION ';';
PTR_DECL -> "ptr" @IDENTIFIER ':' TYPE "=" EXPRESSION ';';
VIEW_DECL -> "view" @IDENTIFIER ":" TYPE "=" EXPRESSION ';';
TRANSFORM_DECL -> "transform" @IDENTIFIER ':' TYPE "=" EXPRESSION ';';

VIEW_STATEMENT -> "view" EXPRESSION "as" @IDENTIFIER ':' TYPE ';';
TRANSFORM_STATEMENT -> "transform" EXPRESSION "to" @IDENTIFIER ':' TYPE ';';

CALL_SIGNATURE -> @IDENTIFIER "(" (EXPRESSION ("," EXPRESSION)*)? ")";
