  3. hardware pointers //IOT/EM

1. reference pointers

fx add(int `a, int `b, int `c) {

  `c = `a + `b; //pointers, deferencing, *(x + 10) = 50
  return `a; //not allowed

}

fx main() {

  let a: int = 10;
  let b: int = 20;
  let c: int = 0;

  add(`a, `b, `c);

  print(c) //= 30

}
